cmake_minimum_required(VERSION 3.20)
project(robotweax_bridge_installed_consumer LANGUAGES CXX)
find_package(RobotweaxBridge 0.1 CONFIG REQUIRED COMPONENTS endpoint)
foreach(component core protocol management journal worker transport endpoint journal_worker)
  if(NOT TARGET RobotweaxBridge::${component})
    message(FATAL_ERROR "Missing installed target ${component}")
  endif()
endforeach()
get_target_property(_worker RobotweaxBridge::journal_worker IMPORTED_LOCATION_RELEASE)
string(FIND "${_worker}" "${EXPECTED_PREFIX}/" _prefix)
if(NOT _prefix EQUAL 0)
  message(FATAL_ERROR "Worker did not relocate with the package")
endif()
add_executable(installed_bridge_consumer main.cpp)
target_link_libraries(installed_bridge_consumer PRIVATE RobotweaxBridge::endpoint)
target_compile_features(installed_bridge_consumer PRIVATE cxx_std_20)
target_compile_options(installed_bridge_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(installed_bridge_consumer PRIVATE WORKER_EXECUTABLE="$<TARGET_FILE:RobotweaxBridge::journal_worker>")
set_target_properties(installed_bridge_consumer PROPERTIES CXX_EXTENSIONS OFF)
