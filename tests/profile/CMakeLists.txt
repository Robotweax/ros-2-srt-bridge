add_executable(profile_probe profile_probe.cpp)
target_compile_features(profile_probe PRIVATE cxx_std_20)
set_target_properties(profile_probe PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(profile_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(profile_probe PRIVATE RobotweaxSRT::srt)
if(BUILD_TESTING)
  foreach(profile control sensor)
    foreach(family ipv4 ipv6)
      foreach(scenario positive backpressure wrong-profile wrong-secret missing-secret missing-gcm missing-sessionauth)
        set(probe_name "profile_${profile}_${family}_${scenario}")
        add_test(NAME ${probe_name} COMMAND profile_probe ${profile} ${family} ${scenario})
        set_tests_properties(${probe_name} PROPERTIES TIMEOUT 12 LABELS "m0;profile;socket")
      endforeach()
      foreach(scenario wrong-profile missing-gcm missing-sessionauth)
        set(probe_name "profile_${profile}_${family}_${scenario}_reverse")
        add_test(NAME ${probe_name} COMMAND profile_probe ${profile} ${family} ${scenario} reverse)
        set_tests_properties(${probe_name} PROPERTIES TIMEOUT 12 LABELS "m0;profile;socket")
      endforeach()
    endforeach()
  endforeach()
endif()
