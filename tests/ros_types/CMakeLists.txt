cmake_minimum_required(VERSION 3.20)
project(robotweax_ros_type_probe LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(ros_type_probe ros_type_probe.cpp)
target_compile_features(ros_type_probe PRIVATE cxx_std_20)
target_compile_options(ros_type_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
ament_target_dependencies(ros_type_probe rclcpp sensor_msgs)
enable_testing()
add_test(NAME ros_type_probe COMMAND ros_type_probe)
set_tests_properties(ros_type_probe PROPERTIES TIMEOUT 15)
