cmake_minimum_required(VERSION 3.20)
project(robotweax_srt_bridge_core VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
add_library(robotweax_srt_bridge_core src/wire.cpp src/clock.cpp src/bandwidth.cpp src/reconnect.cpp)
add_library(RobotweaxBridge::core ALIAS robotweax_srt_bridge_core)
target_include_directories(robotweax_srt_bridge_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(robotweax_srt_bridge_core PUBLIC cxx_std_20)
set_target_properties(robotweax_srt_bridge_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(robotweax_srt_bridge_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
  foreach(contract bandwidth reconnect)
    add_executable(${contract}_tests ../../tests/protocol/${contract}_tests.cpp)
    target_link_libraries(${contract}_tests PRIVATE RobotweaxBridge::core)
    target_compile_options(${contract}_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME ${contract}_contract COMMAND ${contract}_tests)
    set_tests_properties(${contract}_contract PROPERTIES LABELS "m1;${contract}" TIMEOUT 15)
  endforeach()
  add_executable(clock_tests ../../tests/protocol/clock_tests.cpp)
  target_link_libraries(clock_tests PRIVATE RobotweaxBridge::core)
  target_compile_options(clock_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME clock_contract COMMAND clock_tests)
  set_tests_properties(clock_contract PROPERTIES LABELS "m1;clock" TIMEOUT 10)
  add_executable(wire_tests ../../tests/protocol/wire_tests.cpp)
  target_link_libraries(wire_tests PRIVATE RobotweaxBridge::core)
  target_compile_definitions(wire_tests PRIVATE
    GOLDEN_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/protocol/sample-v1.hex")
  target_compile_options(wire_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME wire_contract COMMAND wire_tests)
  set_tests_properties(wire_contract PROPERTIES LABELS "m1;protocol" TIMEOUT 10)
endif()

option(BUILD_MANAGEMENT_CODEC "Build bounded management codec (requires Protobuf)" ON)
if(BUILD_MANAGEMENT_CODEC)
  find_package(Protobuf 3.21 REQUIRED)
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  protobuf_generate_cpp(MANAGEMENT_SOURCES MANAGEMENT_HEADERS
    ${CMAKE_CURRENT_SOURCE_DIR}/../../proto/management_v1.proto
    ${CMAKE_CURRENT_SOURCE_DIR}/../../proto/commands_v1.proto
    ${CMAKE_CURRENT_SOURCE_DIR}/../../proto/errors_v1.proto)
  add_library(robotweax_srt_bridge_management src/management.cpp src/routes.cpp src/session.cpp src/queues.cpp src/commands.cpp src/errors.cpp src/worker/execution_pool.cpp ${MANAGEMENT_SOURCES})
  add_library(RobotweaxBridge::management ALIAS robotweax_srt_bridge_management)
  add_library(RobotweaxBridge::protocol ALIAS robotweax_srt_bridge_management)
  target_include_directories(robotweax_srt_bridge_management PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(robotweax_srt_bridge_management PUBLIC RobotweaxBridge::core PRIVATE protobuf::libprotobuf OpenSSL::Crypto)
  target_compile_options(robotweax_srt_bridge_management PRIVATE -Wall -Wextra -Wpedantic -Werror)
  find_package(SQLite3 3.45 REQUIRED)
  add_library(robotweax_srt_bridge_journal src/journal/journal.cpp src/journal/storage.cpp)
  add_library(RobotweaxBridge::journal ALIAS robotweax_srt_bridge_journal)
  target_link_libraries(robotweax_srt_bridge_journal PUBLIC RobotweaxBridge::protocol PRIVATE SQLite::SQLite3)
  target_compile_options(robotweax_srt_bridge_journal PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Threads REQUIRED)
    add_library(robotweax_srt_bridge_worker src/worker/supervisor.cpp)
    add_library(RobotweaxBridge::worker ALIAS robotweax_srt_bridge_worker)
    target_link_libraries(robotweax_srt_bridge_worker PUBLIC RobotweaxBridge::journal PRIVATE Threads::Threads)
    target_compile_options(robotweax_srt_bridge_worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(robotweax_srt_bridge_worker PROPERTIES CXX_EXTENSIONS OFF)
    add_library(robotweax_srt_bridge_worker_runtime src/worker/runtime.cpp)
    add_library(RobotweaxBridge::worker_runtime ALIAS robotweax_srt_bridge_worker_runtime)
    target_link_libraries(robotweax_srt_bridge_worker_runtime PUBLIC RobotweaxBridge::journal)
    target_compile_options(robotweax_srt_bridge_worker_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(robotweax_srt_bridge_worker_runtime PROPERTIES CXX_EXTENSIONS OFF)
    add_executable(robotweax-journal-worker src/worker/main.cpp)
    target_link_libraries(robotweax-journal-worker PRIVATE RobotweaxBridge::worker_runtime)
    target_compile_options(robotweax-journal-worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(robotweax-journal-worker PROPERTIES CXX_EXTENSIONS OFF)
    if(BUILD_TESTING)
      add_executable(execution_tests ../../tests/protocol/execution_tests.cpp)
      target_link_libraries(execution_tests PRIVATE RobotweaxBridge::protocol Threads::Threads)
      target_compile_options(execution_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
      add_test(NAME execution_contract COMMAND execution_tests)
      set_tests_properties(execution_contract PROPERTIES LABELS "m1;execution;race;process" TIMEOUT 60)
      add_executable(worker_tests ../../tests/protocol/worker_tests.cpp)
      target_link_libraries(worker_tests PRIVATE RobotweaxBridge::worker)
      target_compile_definitions(worker_tests PRIVATE WORKER_EXECUTABLE="$<TARGET_FILE:robotweax-journal-worker>")
      target_compile_options(worker_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
      add_dependencies(worker_tests robotweax-journal-worker)
      add_test(NAME worker_contract COMMAND worker_tests)
      set_tests_properties(worker_contract PROPERTIES LABELS "m1;worker;process;crash;lifecycle" TIMEOUT 90)
    endif()
  endif()
  if(BUILD_TESTING)
    add_executable(journal_quota_tests ../../tests/protocol/journal_quota_tests.cpp)
    target_include_directories(journal_quota_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    target_link_libraries(journal_quota_tests PRIVATE RobotweaxBridge::journal SQLite::SQLite3)
    target_compile_options(journal_quota_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME journal_quota_contract COMMAND journal_quota_tests)
    set_tests_properties(journal_quota_contract PROPERTIES LABELS "m1;journal;quota" TIMEOUT 30)
    add_executable(journal_crash_tests ../../tests/protocol/journal_crash_tests.cpp)
    target_link_libraries(journal_crash_tests PRIVATE RobotweaxBridge::journal)
    target_compile_options(journal_crash_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME journal_crash_contract COMMAND journal_crash_tests)

    add_executable(journal_tests ../../tests/protocol/journal_tests.cpp)
    target_link_libraries(journal_tests PRIVATE RobotweaxBridge::journal)
    target_compile_options(journal_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME journal_contract COMMAND journal_tests)
    set_tests_properties(journal_contract PROPERTIES LABELS "m1;journal" TIMEOUT 120)
  endif()
  if(BUILD_TESTING)
    foreach(contract routes session queues command error command_trace)
      add_executable(${contract}_tests ../../tests/protocol/${contract}_tests.cpp)
      target_link_libraries(${contract}_tests PRIVATE RobotweaxBridge::protocol)
      target_compile_options(${contract}_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
      target_compile_definitions(${contract}_tests PRIVATE GOLDEN_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/protocol")
      add_test(NAME ${contract}_contract COMMAND ${contract}_tests)
      set_tests_properties(${contract}_contract PROPERTIES LABELS "m1;${contract}" TIMEOUT 15)
    endforeach()
    add_executable(management_tests ../../tests/protocol/management_tests.cpp)
    target_link_libraries(management_tests PRIVATE RobotweaxBridge::management protobuf::libprotobuf)
    target_include_directories(management_tests PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_options(management_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME management_contract COMMAND management_tests)
    set_tests_properties(management_contract PROPERTIES LABELS "m1;management" TIMEOUT 15)
  endif()
endif()
