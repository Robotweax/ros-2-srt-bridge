cmake_minimum_required(VERSION 3.20)
project(robotweax_srt_bridge_demo LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(robotweax_srt_bridge_msgs REQUIRED)
find_package(RobotweaxBridge 0.1 CONFIG REQUIRED COMPONENTS protocol)
add_executable(rate_simulator src/simulator.cpp)
target_link_libraries(rate_simulator RobotweaxBridge::protocol OpenSSL::Crypto)
ament_target_dependencies(rate_simulator rclcpp sensor_msgs robotweax_srt_bridge_msgs)
target_compile_options(rate_simulator PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS rate_simulator DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/m3_source_preflight.py scripts/source_payload.py scripts/shutdown_observer.py scripts/scenario_processes.py scripts/endpoint_measurement.py scripts/scenario_files.py scripts/m2_scenario.py scripts/m3_scenario.py scripts/m3_dds_scenario.py scripts/dds_clock.py scripts/dds_command_load.py scripts/command_workload.py scripts/m3_dds_commands_scenario.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(source_trace_test test/source_trace_test.cpp)
  target_include_directories(source_trace_test PRIVATE src)
  target_compile_features(source_trace_test PRIVATE cxx_std_20)
  target_link_libraries(source_trace_test OpenSSL::Crypto)
  ament_target_dependencies(source_trace_test sensor_msgs)
  target_compile_options(source_trace_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  ament_add_gtest(schedule_test test/schedule_test.cpp)
  target_include_directories(schedule_test PRIVATE src)
  target_compile_features(schedule_test PRIVATE cxx_std_20)
  target_compile_options(schedule_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
ament_package()
