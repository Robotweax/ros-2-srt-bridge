cmake_minimum_required(VERSION 3.20)
project(robotweax_srt_bridge_demo LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(robotweax_srt_bridge_msgs REQUIRED)
find_package(RobotweaxBridge 0.1 CONFIG REQUIRED COMPONENTS protocol)
add_executable(rate_simulator src/simulator.cpp)
target_link_libraries(rate_simulator RobotweaxBridge::protocol)
ament_target_dependencies(rate_simulator rclcpp sensor_msgs robotweax_srt_bridge_msgs)
target_compile_options(rate_simulator PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS rate_simulator DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/m2_scenario.py DESTINATION lib/${PROJECT_NAME})
ament_package()
