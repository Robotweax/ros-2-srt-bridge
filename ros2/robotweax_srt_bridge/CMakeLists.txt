cmake_minimum_required(VERSION 3.20)
project(robotweax_srt_bridge LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(robotweax_srt_bridge_msgs REQUIRED)
find_package(RobotweaxBridge 0.1 CONFIG REQUIRED COMPONENTS endpoint worker_runtime)
add_library(sensor_codec INTERFACE)
target_include_directories(sensor_codec INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(sensor_codec INTERFACE cxx_std_20)

add_executable(ros_rate_worker src/ros_worker.cpp)
target_link_libraries(ros_rate_worker sensor_codec RobotweaxBridge::worker_runtime)
ament_target_dependencies(ros_rate_worker rclcpp robotweax_srt_bridge_msgs)
target_compile_options(ros_rate_worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(bridge_node src/bridge_node.cpp)
target_link_libraries(bridge_node sensor_codec RobotweaxBridge::endpoint)
ament_target_dependencies(bridge_node rclcpp sensor_msgs robotweax_srt_bridge_msgs)
target_compile_options(bridge_node PRIVATE -Wall -Wextra -Wpedantic -Werror)
# The node and ROS worker are co-installed in the same libexec directory. The
# default is resolved at runtime from /proc/self/exe, not a build-tree path.
target_compile_definitions(bridge_node PRIVATE ROS_WORKER_EXECUTABLE="ros_rate_worker" SDK_WORKER_EXECUTABLE="$<TARGET_FILE:RobotweaxBridge::journal_worker>")
add_executable(dds_command_node src/dds_command_node.cpp)
target_link_libraries(dds_command_node sensor_codec RobotweaxBridge::journal RobotweaxBridge::protocol OpenSSL::Crypto)
ament_target_dependencies(dds_command_node rclcpp robotweax_srt_bridge_msgs)
target_compile_options(dds_command_node PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS ros_rate_worker bridge_node dds_command_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(dds_expiry_seal_test test/dds_expiry_seal_test.cpp)
  target_link_libraries(dds_expiry_seal_test sensor_codec RobotweaxBridge::journal RobotweaxBridge::protocol)
  ament_add_gtest(dds_command_session_test test/dds_command_session_test.cpp)
  target_link_libraries(dds_command_session_test sensor_codec RobotweaxBridge::protocol OpenSSL::Crypto)
  target_compile_options(dds_command_session_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  ament_add_gtest(sensor_codec_test test/sensor_codec_test.cpp)
  target_link_libraries(sensor_codec_test sensor_codec RobotweaxBridge::protocol)
  ament_target_dependencies(sensor_codec_test rclcpp sensor_msgs)
  target_compile_options(sensor_codec_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
ament_export_include_directories(include)
ament_export_dependencies(rclcpp sensor_msgs robotweax_srt_bridge_msgs OpenSSL)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME} COMPONENT licenses)
ament_package()
