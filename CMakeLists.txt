cmake_minimum_required(VERSION 3.20)
project(robotweax_ros2_srt_bridge VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_subdirectory(src/robotweax_srt_bridge_core)
option(BUILD_PROFILE_PROBE "Build installed SRT profile qualification probe" ON)
option(BUILD_SRT_TRANSPORT "Build single-owner installed-SRT transport" ${BUILD_PROFILE_PROBE})
if(BUILD_PROFILE_PROBE OR BUILD_SRT_TRANSPORT)
  # SRT stays an installed dependency: never add its source tree here.
  find_package(RobotweaxSRT 0.3 CONFIG REQUIRED)
endif()
if(BUILD_SRT_TRANSPORT AND BUILD_MANAGEMENT_CODEC)
  add_subdirectory(src/robotweax_srt_bridge_transport)
endif()
if(BUILD_PROFILE_PROBE)
  add_subdirectory(tests/profile)
  if(BUILD_TESTING AND BUILD_MANAGEMENT_CODEC)
    add_subdirectory(tests/native)
  endif()
endif()

include(cmake/InstallBridge.cmake)
