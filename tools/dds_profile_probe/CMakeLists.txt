cmake_minimum_required(VERSION 3.20)
project(dds_profile_probe LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(robotweax_srt_bridge_msgs REQUIRED)
find_package(rmw_fastrtps_cpp REQUIRED)
find_package(fastrtps REQUIRED)
add_executable(dds_profile_probe main.cpp)
target_compile_features(dds_profile_probe PRIVATE cxx_std_20)
ament_target_dependencies(dds_profile_probe rclcpp sensor_msgs robotweax_srt_bridge_msgs rmw_fastrtps_cpp)
target_link_libraries(dds_profile_probe fastrtps)
target_compile_options(dds_profile_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
